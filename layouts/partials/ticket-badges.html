{{- /* Turns ticket keys (EUR-42, optionally followed by ":") in rendered HTML into badges. Usage: partial "ticket-badges.html" $html An optional type right after the key highlights the badge: EUR-1[epic], EUR-7[feature], EUR-9[bugfix]. Keys inside attributes or URLs (preceded by "/", "=", quotes …) stay untouched. With params.jira.url set, every badge links to /browse/. */ -}} {{- $key := site.Params.jira.projectKey | default "EUR" -}} {{- $re := printf `(^|[\s>(\[])(%s-\d+)\b(?:\[(epic|feature|bugfix)\])?:?` $key -}} {{- $badge := `$1$2` -}} {{- with site.Params.jira.url -}} {{- $badge = printf `$1$2` (strings.TrimSuffix "/" .) -}} {{- end -}} {{- return safeHTML (replaceRE $re $badge (string .)) -}}
